async function sendAppMsg() { const inputEl = document.getElementById("chat-input"); const text = inputEl.value.trim(); if (!text) return; appendMessage("user", text); inputEl.value = ""; chatHistory.push({ role: "user", content: text }); try { const response = await fetch("https://api.groq.com/openai/v1/chat/completions", { method: "POST", headers: { "Authorization": `Bearer ${GROQ_API_KEY}`, "Content-Type": "application/json" }, body: JSON.stringify({ model: "llama-3.1-8b-instant", messages: [ { role: "system", content: SYSTEM_INSTRUCTION }, ...chatHistory ] }) }); const data = await response.json(); if (data.error) { throw new Error(data.error.message || "Error al conectar con la API de Groq"); } const botReply = data.choices[0].message.content; chatHistory.push({ role: "assistant", content: botReply }); appendMessage("model", botReply); } catch (err) { appendMessage("model", "AVISO: " + err.message); } }
